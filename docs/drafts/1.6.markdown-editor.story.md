```typescript
// packages/ui/src/composables/useMarkdownEditor.ts
export interface EditorConfig {
  theme: 'light' | 'dark';
  language: string;
  toolbars: string[];
  codeTheme: string;
  previewTheme: string;
}

export const defaultToolbarConfig = [
  'bold', 'italic', 'strikeThrough', '|',
  'title', 'sub', 'sup', 'quote', 'unorderedList', 'orderedList', 'task', '|',
  'codeRow', 'code', 'link', 'image', 'table', 'mermaid', '|',
  'revoke', 'next', 'save', '|',
  '=',
  'pageFullscreen', 'fullscreen', 'preview', 'htmlPreview', 'catalog'
];

export function useMarkdownEditor() {
  const { services } = useServices();
  const { currentTheme } = useTheme();
  const { t, locale } = useI18n();
  
  // Reactive state
  const content = ref('');
  const originalContent = ref('');
  const currentFile = ref<FileInfo | null>(null);
  const viewMode = ref<'edit' | 'preview' | 'split'>('edit');
  const loading = ref(false);
  
  // Computed properties
  const hasUnsavedChanges = computed(() => {
    return content.value !== originalContent.value;
  });
  
  const editorTheme = computed(() => {
    return currentTheme.value === 'dark' ? 'dark' : 'light';
  });
  
  const currentFileName = computed(() => {
    return currentFile.value?.name || t('editor.untitled');
  });
  
  // File operations
  const loadFile = async (file: FileInfo) => {
    loading.value = true;
    try {
      const fileContent = await services.value?.webdavService.getFile(file.path);
      content.value = fileContent || '';
      originalContent.value = content.value;
      currentFile.value = file;
    } catch (error) {
      console.error('Failed to load file:', error);
      showNotification(t('editor.loadError'), 'error');
    } finally {
      loading.value = false;
    }
  };
  
  return {
    content,
    currentFileName,
    hasUnsavedChanges,
    viewMode,
    editorTheme,
    loadFile,
    /* other methods */
  };
}
```

### Performance Optimization Configuration
[Source: large file handling requirements]
```typescript
export const editorPerformanceConfig = {
  // Large file handling
  maxFileSize: 10 * 1024 * 1024, // 10MB
  chunkSize: 64 * 1024, // 64KB chunks for progressive loading
  
  // Rendering optimization
  enableVirtualization: true,
  virtualThreshold: 1000, // lines
  
  // Change detection
  debounceDelay: 300, // ms
  throttleDelay: 16, // ~60fps
  
  // Syntax highlighting
  enableWorkerHighlighting: true,
  highlightingTimeout: 5000, // ms
  
  // History management
  maxHistorySize: 100,
  historyMergeDelay: 1000, // ms
};

// Progressive file loading
export async function loadLargeFile(
  filePath: string, 
  webdavService: any,
  onProgress?: (progress: number) => void
): Promise<string> {
  const fileInfo = await webdavService.getFileInfo(filePath);
  const fileSize = fileInfo.size;
  
  if (fileSize <= editorPerformanceConfig.chunkSize) {
    return await webdavService.getFile(filePath);
  }
  
  // Load in chunks for large files
  let content = '';
  let loaded = 0;
  const totalChunks = Math.ceil(fileSize / editorPerformanceConfig.chunkSize);
  
  for (let i = 0; i < totalChunks; i++) {
    const start = i * editorPerformanceConfig.chunkSize;
    const end = Math.min(start + editorPerformanceConfig.chunkSize, fileSize);
    
    const chunk = await webdavService.getFileRange(filePath, start, end);
    content += chunk;
    loaded += chunk.length;
    
    onProgress?.(loaded / fileSize);
  }
  
  return content;
}
```

### Error Handling Strategy
[Source: existing error patterns]
```typescript
export const editorErrorHandler = {
  handleLoadError: (error: Error, filePath: string) => {
    let message = 'Failed to load file';
    
    if (error.name === 'NetworkError') {
      message = 'Network error - check connection';
    } else if (error.name === 'PermissionError') {
      message = 'Permission denied - check access rights';
    } else if (error.message.includes('too large')) {
      message = 'File too large to load';
    }
    
    showNotification(message, 'error', 0, [{
      label: 'Retry',
      action: () => retryLoadFile(filePath)
    }]);
  },
  
  handleSaveError: (error: Error, filePath: string) => {
    showNotification('Failed to save file', 'error', 0, [{
      label: 'Retry',
      action: () => retrySaveFile(filePath)
    }]);
  }
};
```
