### File Operations Service Integration
[Source: core services structure]
```typescript
// packages/ui/src/composables/useFileOperations.ts
export interface FileOperationsHandler {
  async createFile(parentPath: string, name: string): Promise<void>;
  async createFolder(parentPath: string, name: string): Promise<void>;
  async renameItem(oldPath: string, newName: string): Promise<void>;
  async deleteItem(path: string): Promise<void>;
  async copyPath(path: string): Promise<void>;
}

export function useFileOperations(services: Ref<AppServices | null>) {
  const { showNotification } = useToast();
  const fileTreeStore = useFileTreeStore();
  
  const createFile = async (parentPath: string, name: string) => {
    try {
      const fullPath = `${parentPath}/${name}.md`;
      await services.value?.webdavService.putFile(fullPath, '# ' + name);
      await fileTreeStore.refreshNode(parentPath);
      showNotification('File created successfully', 'success');
    } catch (error) {
      showNotification('Failed to create file', 'error');
      throw error;
    }
  };
  
  return { createFile, /* other methods */ };
}
```

### Filename Validation
[Source: existing validation patterns]
```typescript
export const validateFilename = (filename: string): string | null => {
  if (!filename.trim()) {
    return 'Filename cannot be empty';
  }
  
  if (filename.length > 255) {
    return 'Filename too long (max 255 characters)';
  }
  
  // Allow alphanumeric, dash, underscore, and space
  if (!/^[a-zA-Z0-9\s\-_]+$/.test(filename)) {
    return 'Filename contains invalid characters';
  }
  
  // Reserved names
  const reserved = ['CON', 'PRN', 'AUX', 'NUL', 'COM1', 'COM2', 'COM3', 'COM4', 'COM5', 'COM6', 'COM7', 'COM8', 'COM9', 'LPT1', 'LPT2', 'LPT3', 'LPT4', 'LPT5', 'LPT6', 'LPT7', 'LPT8', 'LPT9'];
  if (reserved.includes(filename.toUpperCase())) {
    return 'Filename is reserved';
  }
  
  return null;
};

export const sanitizeFilename = (filename: string): string => {
  return filename
    .trim()
    .replace(/[<>:"/\\|?*]/g, '') // Remove invalid chars
    .replace(/\s+/g, '-')         // Replace spaces with dashes
    .toLowerCase();
};
```

### Keyboard Event Handling
[Source: existing keyboard navigation patterns]
```typescript
export function useKeyboardShortcuts(fileTreeStore: any, operations: any) {
  const handleKeydown = (event: KeyboardEvent) => {
    if (!fileTreeStore.selectedNode) return;
    
    switch (event.key) {
      case 'Delete':
        event.preventDefault();
        operations.confirmDelete(fileTreeStore.selectedNode);
        break;
      case 'F2':
        event.preventDefault();
        operations.startRename(fileTreeStore.selectedNode);
        break;
      case 'Escape':
        if (operations.showContextMenu) {
          operations.hideContextMenu();
        }
        break;
    }
  };
  
  onMounted(() => {
    document.addEventListener('keydown', handleKeydown);
  });
  
  onUnmounted(() => {
    document.removeEventListener('keydown', handleKeydown);
  });
}
```

```typescript
const handleOperationError = (error: Error, operation: string) => {
  let message = `Failed to ${operation}`;
  let actions = [];
  
  if (error.name === 'NetworkError') {
    message += ' - Check your connection';
    actions.push({
      label: 'Retry',
      action: () => retryLastOperation()
    });
  } else if (error.name === 'PermissionError') {
    message += ' - Insufficient permissions';
  }
  
  showNotification(message, 'error', 0, actions);
};
```
