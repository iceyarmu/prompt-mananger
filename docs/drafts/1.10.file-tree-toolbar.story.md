# Story 1.10: Add File Tree Toolbar and Status

## Status
Draft

## Story
**As a** user,
**I want** quick access to file tree actions and connection status,
**so that** I can manage my WebDAV connection efficiently.

## Acceptance Criteria
1. Add toolbar at bottom of file tree panel with sticky positioning and proper z-index
2. Include "Configure" button that opens existing WebDAV settings modal seamlessly
3. Include "Refresh" button that reloads file tree from server with loading animation
4. Display connection status indicator (green=connected, red=disconnected, yellow=connecting) with smooth transitions
5. Show current connection profile name when connected with truncation for long names
6. Provide tooltip with connection details on hover including URL and last sync time
7. Toolbar integrates seamlessly with file tree component without layout breaks
8. Configure button opens existing WebDAV configuration modal using current patterns
9. Refresh triggers WebDAVService.listFolder() for root with proper error handling
10. Status updates automatically on connection changes with 500ms max latency
11. Visual feedback for all operations using existing loading spinner patterns
12. Status updates within 500ms of state change with debounced updates
13. Refresh completes within 3 seconds for typical tree with timeout handling
14. Toolbar remains visible during tree scrolling with proper sticky behavior
15. Responsive design for narrow sidebars (200px-400px) with graceful degradation
16. Accessible with keyboard navigation and screen reader support

```vue
<script setup lang="ts">
import { computed, ref } from 'vue';
import { useWebDAVStore } from '../composables/useWebDAVStore';
import { useFileTreeStore } from '../composables/useFileTreeStore';
import { useToast } from '../composables/useToast';
import { useI18n } from 'vue-i18n';

const webdavStore = useWebDAVStore();
const fileTreeStore = useFileTreeStore();
const { showNotification } = useToast();
const { t } = useI18n();

const refreshing = ref(false);

// Computed properties
const connectionStatus = computed(() => webdavStore.connectionStatus);
const activeProfile = computed(() => webdavStore.activeProfile);
const isConnected = computed(() => connectionStatus.value === 'connected');
const lastSyncTime = computed(() => webdavStore.lastSyncTime);

const refreshTree = async () => {
  if (!isConnected.value) return;
  
  refreshing.value = true;
  try {
    await fileTreeStore.loadTree();
    webdavStore.updateLastSync();
    showNotification(t('fileTree.refresh.success'), 'success', 2000);
  } catch (error) {
    console.error('Failed to refresh tree:', error);
    showNotification(t('fileTree.refresh.error'), 'error');
  } finally {
    refreshing.value = false;
  }
};

const openConfiguration = () => {
  webdavStore.showConfigModal = true;
};
</script>
```

```vue
<template>
  <div class="connection-status flex items-center gap-2 min-w-0">

    <div class="relative flex-shrink-0">
      <div
        class="status-dot w-2 h-2 rounded-full transition-colors duration-300"
        :class="statusDotClass"

        :class="statusDotClass"
      />
    </div>
    
    <!-- Status Text with Tooltip -->
    <TooltipProvider>
      <Tooltip>
        <TooltipTrigger as-child>
          <div class="min-w-0 flex-1">
            <div class="text-xs font-medium truncate">
              <template v-if="profile && status === 'connected'">
                {{ profile.name }}
              </template>
              <template v-else-if="status === 'connecting'">
                {{ t('webdav.status.connecting') }}
              </template>
              <template v-else-if="status === 'error'">
                {{ t('webdav.status.error') }}
              </template>
              <template v-else>
                {{ t('webdav.status.disconnected') }}
              </template>
            </div>
            
            <!-- Connection quality indicator -->
            <div v-if="profile && status === 'connected'" class="flex items-center gap-1">
              <div class="w-1 h-2 bg-green-400 rounded-sm" />
              <div class="w-1 h-2 bg-green-400 rounded-sm" :class="{ 'bg-gray-300': connectionQuality < 2 }" />
              <div class="w-1 h-2 bg-green-400 rounded-sm" :class="{ 'bg-gray-300': connectionQuality < 3 }" />
            </div>
          </div>
        </TooltipTrigger>
        
        <TooltipContent v-if="profile" class="max-w-64">
          <div class="space-y-1 text-xs">
            <div class="font-medium">{{ profile.name }}</div>
            <div class="text-gray-400">{{ profile.url }}</div>
            <div v-if="lastSync" class="text-gray-400">
              {{ t('webdav.status.lastSync') }}: {{ formatRelativeTime(lastSync) }}
            </div>
            <div class="text-gray-400">
              {{ t('webdav.status.status') }}: {{ t(`webdav.status.${status}`) }}
            </div>
          </div>
        </TooltipContent>
        
        <TooltipContent v-else>
          <div class="text-xs">
            {{ t(`webdav.status.${status}`) }}
          </div>
        </TooltipContent>
      </Tooltip>
    </TooltipProvider>
  </div>
</template>

<script setup lang="ts">
import { computed } from 'vue';

interface WebDAVProfile {
  name: string;
  url: string;
  lastSync?: Date;
}

interface Props {
  status: 'connected' | 'disconnected' | 'connecting' | 'error';
  profile?: WebDAVProfile | null;
  lastSync?: Date;
}

const props = defineProps<Props>();

const statusDotClass = computed(() => {
  switch (props.status) {
    case 'connected':
      return 'bg-green-500';
    case 'connecting':
      return 'bg-yellow-500';
    case 'error':
      return 'bg-red-500';
    default:
      return 'bg-gray-400';
  }
});

const connectionQuality = computed(() => {
  // Mock connection quality based on last sync time
  if (!props.lastSync) return 1;
  const timeSinceSync = Date.now() - props.lastSync.getTime();
  if (timeSinceSync < 60000) return 3; // < 1 minute = excellent
  if (timeSinceSync < 300000) return 2; // < 5 minutes = good
  return 1; // > 5 minutes = poor
});

const formatRelativeTime = (date: Date) => {
  const now = new Date();
  const diffMs = now.getTime() - date.getTime();
  const diffMins = Math.floor(diffMs / 60000);
  
  if (diffMins < 1) return t('time.justNow');
  if (diffMins < 60) return t('time.minutesAgo', { count: diffMins });
  
  const diffHours = Math.floor(diffMins / 60);
  if (diffHours < 24) return t('time.hoursAgo', { count: diffHours });
  
  return date.toLocaleDateString();
};
</script>

  box-shadow: 0 0 0 1px rgba(0, 0, 0, 0.1);
}
</style>
```

### WebDAV Store Integration
[Source: existing store patterns and connection management]
```typescript
// packages/ui/src/composables/useWebDAVStore.ts (additions)
export function useWebDAVStore() {
  const connectionStatus = ref<'connected' | 'disconnected' | 'connecting' | 'error'>('disconnected');
  const activeProfile = ref<WebDAVProfile | null>(null);
  const lastSyncTime = ref<Date | null>(null);
  const showConfigModal = ref(false);
  
  // Connection monitoring
  const connectionMonitorInterval = ref<NodeJS.Timeout | null>(null);
  
  const startConnectionMonitoring = () => {
    stopConnectionMonitoring();
    
    connectionMonitorInterval.value = setInterval(async () => {
      if (activeProfile.value && connectionStatus.value === 'connected') {
        try {
          await testConnection(activeProfile.value);
        } catch (error) {
          console.warn('Connection test failed:', error);
          connectionStatus.value = 'error';
        }
      }
    }, 30000); // Check every 30 seconds
  };
  
  const stopConnectionMonitoring = () => {
    if (connectionMonitorInterval.value) {
      clearInterval(connectionMonitorInterval.value);
      connectionMonitorInterval.value = null;
    }
  };
  
  const connect = async (profile: WebDAVProfile) => {
    connectionStatus.value = 'connecting';
    
    try {
      // Use existing WebDAV service connection logic
      await services.value?.webdavService.connect(profile);
      
      activeProfile.value = profile;
      connectionStatus.value = 'connected';
      lastSyncTime.value = new Date();
      
      startConnectionMonitoring();
      
    } catch (error) {
      connectionStatus.value = 'error';
      throw error;
    }
  };
  
  const disconnect = () => {
    stopConnectionMonitoring();
    connectionStatus.value = 'disconnected';
    activeProfile.value = null;
    lastSyncTime.value = null;
  };
  
  const updateLastSync = () => {
    lastSyncTime.value = new Date();
  };
  
  const testConnection = async (profile?: WebDAVProfile) => {
    const targetProfile = profile || activeProfile.value;
    if (!targetProfile) throw new Error('No profile to test');
    
    // Use existing WebDAV service test method
    await services.value?.webdavService.testConnection(targetProfile);
  };
  
  return {
    connectionStatus: readonly(connectionStatus),
    activeProfile: readonly(activeProfile),
    lastSyncTime: readonly(lastSyncTime),
    showConfigModal,
    connect,
    disconnect,
    updateLastSync,
    testConnection,
    startConnectionMonitoring,
    stopConnectionMonitoring
  };
}
```

### Error Handling and Recovery
[Source: existing error patterns]
```typescript
// packages/ui/src/services/ConnectionErrorHandler.ts
export class ConnectionErrorHandler {
  handleRefreshError(error: Error): void {
    let message = 'Failed to refresh file tree';
    let actions: Array<{ label: string; action: () => void }> = [];
    
    if (error.name === 'NetworkError') {
      message = 'Network error - check connection';
      actions.push({
        label: 'Retry',
        action: () => this.retryRefresh()
      });
    } else if (error.message.includes('timeout')) {
      message = 'Request timed out';
      actions.push({
        label: 'Retry',
        action: () => this.retryRefresh()
      });
    } else if (error.message.includes('unauthorized')) {
      message = 'Authentication failed - check credentials';
      actions.push({
        label: 'Configure',
        action: () => this.openConfiguration()
      });
    }
    
    showNotification(message, 'error', 0, actions);
  }
  
  handleConnectionError(error: Error): void {
    const webdavStore = useWebDAVStore();
    
    webdavStore.connectionStatus = 'error';
    
    let message = 'Connection failed';
    if (error.message.includes('network')) {
      message = 'Network unreachable';
    } else if (error.message.includes('timeout')) {
      message = 'Connection timed out';
    }
    
    showNotification(message, 'error', 5000);
  }
}
```
