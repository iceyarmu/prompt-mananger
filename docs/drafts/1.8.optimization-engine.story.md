# Story 1.8: Connect Optimization Engine to Editor

## Status
Draft

## Story
**As a** user,
**I want** to optimize my prompts directly from the editor,
**so that** I can improve them without switching contexts.

## Acceptance Criteria
1. Add "Optimize" button to editor toolbar with proper styling and positioning
2. Button sends current editor content to existing optimization engine without modifications
3. Display optimization results in modal with before/after comparison view
4. Show split view and diff view for result comparison with syntax highlighting
5. Provide "Apply" button to replace content with optimized version seamlessly
6. Preserve original version for undo capability with history management
7. Use existing optimization service without modifications to maintain compatibility
8. Optimization API calls follow existing patterns with proper error handling
9. Loading states consistent with current optimization UI using existing components
10. Error handling uses existing notification system with retry options
11. Performance characteristics unchanged from current implementation
12. Optimization completes within current time limits with timeout handling
13. UI remains responsive during optimization with proper async handling
14. Clear indication of what will be optimized with content preview
15. Diff view highlights changes clearly using existing diff component
16. Original content recoverable via undo with proper state management

### Existing Optimization Service Integration
[Source: IPromptService types and existing optimization patterns]
```typescript
// packages/ui/src/services/EditorOptimizationHandler.ts
import { IPromptService, OptimizationRequest } from '@prompt-optimizer/core';
import type { AppServices } from '../types/services';

export class EditorOptimizationHandler {
  private originalContent: string = '';
  
  constructor(
    private services: AppServices,
    private editorStore: any
  ) {}
  
  async optimizeCurrentContent(): Promise<OptimizationResult> {
    const content = this.editorStore.content;
    this.originalContent = content;
    
    // Use existing optimization service exactly as current implementation
    const optimizationRequest: OptimizationRequest = {
      optimizationMode: this.services.optimizationStore.mode, // 'system' | 'user'
      targetPrompt: content,
      templateId: this.services.optimizationStore.selectedTemplate,
      modelKey: this.services.optimizationStore.selectedModel
    };
    
    const result = await this.services.promptService.optimizePrompt(optimizationRequest);
    
    return {
      original: this.originalContent,
      optimized: result,
      metadata: {
        model: optimizationRequest.modelKey,
        template: optimizationRequest.templateId,
        mode: optimizationRequest.optimizationMode,
        timestamp: new Date(),
        originalLength: content.length,
        optimizedLength: result.length,
        changePercentage: Math.round(((result.length - content.length) / content.length) * 100)
      }
    };
  }
  
  applyOptimization(optimizedContent: string): void {
    // Store original for undo
    this.editorStore.pushToUndoStack(this.originalContent);
    
    // Apply optimized content
    this.editorStore.updateContent(optimizedContent);
    this.editorStore.markAsModified();
    
    // Show success notification
    this.services.notificationService.show('Optimization applied', 'success');
  }
  
  revertOptimization(): void {
    this.editorStore.updateContent(this.originalContent);
    this.services.notificationService.show('Reverted to original', 'info');
  }
}
```

import { computed, ref } from 'vue';
import { BaseModal, Button, Icon, Tabs, TabsList, TabsTrigger, TabsContent } from '../index';

interface OptimizationResult {
  original: string;
  optimized: string;
  metadata: {
    model: string;
    template?: string;
    mode: string;
    timestamp: Date;
    originalLength: number;
    optimizedLength: number;
    changePercentage: number;
  };
}

interface Props {
  result: OptimizationResult | null;
  visible: boolean;
}

const props = defineProps<Props>();
const emit = defineEmits<{
  'update:visible': [value: boolean];
  'apply': [content: string];
}>();

const currentView = ref('split');
const applying = ref(false);

const changeColorClass = computed(() => {
  const change = props.result?.metadata.changePercentage || 0;
  if (change > 0) return 'text-red-600';
  if (change < 0) return 'text-green-600';
  return 'text-gray-600';
});

const applyOptimization = async () => {
  if (!props.result) return;
  
  applying.value = true;
  try {
    emit('apply', props.result.optimized);
    closeModal();
  } finally {
    applying.value = false;
  }
};

const closeModal = () => {
  emit('update:visible', false);
};

const copyToClipboard = async (text: string) => {
  try {
    await navigator.clipboard.writeText(text);
    // Use existing notification service
    showNotification('Copied to clipboard', 'success');
  } catch (error) {
    showNotification('Failed to copy', 'error');
  }
};

### Integration with Existing Optimization Store
[Source: existing optimization patterns]
```typescript
// packages/ui/src/composables/useEditorOptimization.ts
export function useEditorOptimization(services: Ref<AppServices | null>) {
  const optimizing = ref(false);
  const showResultsModal = ref(false);
  const optimizationResult = ref<OptimizationResult | null>(null);
  
  const optimizeContent = async (content: string) => {
    if (!content.trim() || !services.value) {
      showNotification('No content to optimize', 'warning');
      return;
    }
    
    optimizing.value = true;
    
    try {
      const handler = new EditorOptimizationHandler(services.value, editorStore);
      const result = await handler.optimizeCurrentContent();
      
      optimizationResult.value = result;
      showResultsModal.value = true;
      
    } catch (error) {
      console.error('Optimization failed:', error);
      showNotification(
        error.message || 'Optimization failed', 
        'error',
        0,
        [{
          label: 'Retry',
          action: () => optimizeContent(content)
        }]
      );
    } finally {
      optimizing.value = false;
    }
  };
  
  const applyOptimization = async (optimizedContent: string) => {
    if (!services.value) return;
    
    const handler = new EditorOptimizationHandler(services.value, editorStore);
    handler.applyOptimization(optimizedContent);
    
    showResultsModal.value = false;
    optimizationResult.value = null;
  };
  
  return {
    optimizing: readonly(optimizing),
    showResultsModal,
    optimizationResult,
    optimizeContent,
    applyOptimization
  };
}
```

// Monitor optimization performance to ensure no regression
export const optimizationMetrics = {
  trackOptimization: (startTime: number, endTime: number, contentLength: number) => {
    const duration = endTime - startTime;
    const performance = {
      duration,
      contentLength,
      throughput: contentLength / duration,
      timestamp: new Date()
    };
    
    // Log if performance degrades
    if (duration > 10000) { // 10 seconds
      console.warn('Slow optimization detected:', performance);
    }
    
    // Store metrics for analysis
    analytics?.track('optimization_performance', performance);
  }
};

### Error Handling Strategy
[Source: existing error patterns]
```typescript
export const optimizationErrorHandler = {
  handleOptimizationError: (error: Error, context: string) => {
    let message = 'Optimization failed';
    let actions: Array<{ label: string; action: () => void }> = [];
    
    if (error.name === 'NetworkError') {
      message = 'Network error during optimization';
      actions.push({
        label: 'Retry',
        action: () => retryOptimization()
      });
    } else if (error.message.includes('rate limit')) {
      message = 'Rate limit exceeded - please wait';
    } else if (error.message.includes('content too large')) {
      message = 'Content too large for optimization';
    }
    
    showNotification(message, 'error', 0, actions);
    
    // Log for debugging
    console.error(`Optimization error in ${context}:`, error);
  }
};
```
