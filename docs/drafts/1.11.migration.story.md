# Story 1.11: Implement Migration and Backward Compatibility

## Status
Removed

## Story
**As a** user,
**I want** my existing local prompts migrated to the new system,
**so that** I don't lose my previous work.

## Acceptance Criteria
1. Detect existing prompts in local storage on first launch with version checking
2. Prompt user to migrate with clear explanation of benefits and process
3. Create backup of local storage before migration with downloadable JSON export
4. Convert each local prompt to .md file format with proper metadata preservation
5. Upload converted files to WebDAV storage using existing service patterns
6. Verify successful migration before clearing local storage with integrity checks
7. Provide rollback option if migration fails with complete restoration capability
8. Migration runs only once per browser/user with persistent completion tracking
9. Existing app remains functional if migration skipped with graceful fallback
10. Migration process doesn't block UI with proper async handling and progress indication
11. Progress indication during migration with step-by-step feedback and cancellation option
12. Local storage remains until migration confirmed with user verification
13. Zero data loss during migration process with atomic operations and validation
14. Migration handles interruptions gracefully with resume capability
15. Clear error messages for migration failures with specific troubleshooting steps
16. Backup downloadable as JSON file with complete data export including metadata

```typescript
// packages/ui/src/services/MigrationManager.ts
interface LegacyPrompt {
  id: string;
  title: string;
  content: string;
  created: string | Date;
  modified: string | Date;
  tags?: string[];
  category?: string;
  version?: string;
  metadata?: Record<string, any>;
}

interface MigrationState {
  status: 'not_started' | 'in_progress' | 'completed' | 'failed' | 'cancelled';
  startTime?: Date;
  endTime?: Date;
  totalItems: number;
  processedItems: number;
  failedItems: number;
  currentStep: string;
  backupPath?: string;
  migrationId: string;
}

export class MigrationManager {
  private readonly MIGRATION_KEY = 'prompt_migration_v2_state';
  private readonly LEGACY_KEYS = [
    'saved_prompts',           // v1.0 format
    'prompt_data',             // v1.1 format
    'user_prompts',            // v1.2 format
    'optimization_history'     // optimization history
  ];
  
  constructor(
    private storageService: any,
    private webdavService: any,
    private backupManager: BackupManager
  ) {}
  
  async checkMigrationNeeded(): Promise<{
    needed: boolean;
    dataFound: LegacyPrompt[];
    estimatedTime: number;
  }> {
    // Check if migration already completed
    const migrationState = await this.getMigrationState();
    if (migrationState?.status === 'completed') {
      return { needed: false, dataFound: [], estimatedTime: 0 };
    }
    
    // Detect legacy data across all known keys
    const legacyData: LegacyPrompt[] = [];
    
    for (const key of this.LEGACY_KEYS) {
      const data = await this.storageService.getItem(key);
      if (data) {
        try {
          const parsed = JSON.parse(data);
          const prompts = Array.isArray(parsed) ? parsed : [parsed];
          legacyData.push(...this.normalizeLegacyPrompts(prompts));
        } catch (error) {
          console.warn(`Failed to parse legacy data from ${key}:`, error);
        }
      }
    }
    
    // Remove duplicates based on content hash
    const uniquePrompts = this.deduplicatePrompts(legacyData);
    
    return {
      needed: uniquePrompts.length > 0,
      dataFound: uniquePrompts,
      estimatedTime: this.estimateMigrationTime(uniquePrompts.length)
    };
  }
  
  private normalizeLegacyPrompts(rawData: any[]): LegacyPrompt[] {
    return rawData.map((item, index) => ({
      id: item.id || `legacy_${index}_${Date.now()}`,
      title: item.title || item.name || `Untitled ${index + 1}`,
      content: item.content || item.text || '',
      created: new Date(item.created || item.createdAt || Date.now()),
      modified: new Date(item.modified || item.updatedAt || item.created || Date.now()),
      tags: item.tags || [],
      category: item.category || 'migrated',
      version: item.version || '1.0',
      metadata: {
        originalId: item.id,
        migrationSource: 'localStorage',
        ...item.metadata
      }
    }));
  }
  
  private deduplicatePrompts(prompts: LegacyPrompt[]): LegacyPrompt[] {
    const seen = new Set<string>();
    return prompts.filter(prompt => {
      const hash = this.generateContentHash(prompt.content + prompt.title);
      if (seen.has(hash)) {
        return false;
      }
      seen.add(hash);
      return true;
    });
  }
  
  private generateContentHash(content: string): string {
    // Simple hash function for deduplication
    let hash = 0;
    for (let i = 0; i < content.length; i++) {
      const char = content.charCodeAt(i);
      hash = ((hash << 5) - hash) + char;
      hash = hash & hash; // Convert to 32-bit integer
    }
    return hash.toString();
  }
  
  private estimateMigrationTime(promptCount: number): number {
    // Estimate: ~100ms per prompt + overhead
    return Math.max(promptCount * 100 + 2000, 5000);
  }
}
```

### Data Conversion and Upload Service
[Source: existing WebDAV patterns and markdown conversion]
```typescript
// packages/ui/src/services/MigrationConverter.ts
export class MigrationConverter {
  constructor(
    private webdavService: any,
    private progressCallback?: (progress: number, operation: string) => void
  ) {}
  
  async convertAndUpload(
    prompts: LegacyPrompt[],
    migrationFolder: string = '/migrated'
  ): Promise<MigrationResult> {
    const results: MigrationResult = {
      successful: [],
      failed: [],
      totalProcessed: 0
    };
    
    // Create migration folder structure
    try {
      await this.webdavService.createFolder(migrationFolder);
      await this.webdavService.createFolder(`${migrationFolder}/categories`);
    } catch (error) {
      console.warn('Failed to create migration folders:', error);
    }
    
    // Convert and upload in batches for better performance
    const batchSize = 5;
    for (let i = 0; i < prompts.length; i += batchSize) {
      const batch = prompts.slice(i, i + batchSize);
      
      await Promise.allSettled(
        batch.map(async (prompt) => {
          try {
            const convertedContent = this.convertToMarkdown(prompt);
            const filePath = this.generateFilePath(prompt, migrationFolder);
            
            this.progressCallback?.(
              (i + batch.indexOf(prompt)) / prompts.length * 100,
              `Uploading ${prompt.title}...`
            );
            
            await this.webdavService.putFile(filePath, convertedContent);
            
            // Verify upload
            const uploaded = await this.webdavService.getFile(filePath);
            if (uploaded !== convertedContent) {
              throw new Error('Upload verification failed');
            }
            
            results.successful.push({
              original: prompt,
              filePath,
              size: convertedContent.length
            });
            
          } catch (error) {
            console.error(`Failed to migrate prompt "${prompt.title}":`, error);
            results.failed.push({
              prompt,
              error: error.message,
              retryable: this.isRetryableError(error)
            });
          } finally {
            results.totalProcessed++;
          }
        })
      );
      
      // Brief pause between batches to prevent overwhelming the server
      await new Promise(resolve => setTimeout(resolve, 100));
    }
    
    return results;
  }
  
  private convertToMarkdown(prompt: LegacyPrompt): string {
    const frontmatter = this.generateFrontmatter(prompt);
    const content = this.processContent(prompt.content);
    
    return `---
${frontmatter}
---

# ${prompt.title}

${content}

---

## Migration Information
- **Original ID**: ${prompt.id}
- **Created**: ${new Date(prompt.created).toISOString()}
- **Modified**: ${new Date(prompt.modified).toISOString()}
- **Migrated**: ${new Date().toISOString()}
- **Source**: Local Storage Migration
${prompt.tags?.length ? `- **Tags**: ${prompt.tags.join(', ')}` : ''}
${prompt.category ? `- **Category**: ${prompt.category}` : ''}
`;
  }
  
  private generateFrontmatter(prompt: LegacyPrompt): string {
    const metadata = {
      title: prompt.title,
      created: new Date(prompt.created).toISOString(),
      modified: new Date(prompt.modified).toISOString(),
      migrated: new Date().toISOString(),
      tags: prompt.tags || [],
      category: prompt.category || 'migrated',
      version: prompt.version || '1.0',
      id: prompt.id,
      ...prompt.metadata
    };
    
    return Object.entries(metadata)
      .map(([key, value]) => {
        if (Array.isArray(value)) {
          return `${key}: [${value.map(v => `"${v}"`).join(', ')}]`;
        }
        return `${key}: "${value}"`;
      })
      .join('\n');
  }
  
  private generateFilePath(prompt: LegacyPrompt, basePath: string): string {
    const sanitizedTitle = this.sanitizeFilename(prompt.title);
    const category = prompt.category ? this.sanitizeFilename(prompt.category) : 'general';
    
    // Create category-based folder structure
    const categoryPath = `${basePath}/categories/${category}`;
    const filename = `${sanitizedTitle}.md`;
    
    return `${categoryPath}/${filename}`;
  }
  
  private sanitizeFilename(name: string): string {
    return name
      .replace(/[<>:"/\\|?*]/g, '')  // Remove invalid chars
      .replace(/\s+/g, '-')          // Replace spaces with dashes
      .replace(/--+/g, '-')          // Replace multiple dashes with single
      .replace(/^-|-$/g, '')         // Remove leading/trailing dashes
      .toLowerCase()
      .substring(0, 100);            // Limit length
  }
  
  private processContent(content: string): string {
    // Clean up content for markdown
    return content
      .replace(/\r\n/g, '\n')        // Normalize line endings
      .replace(/\n{3,}/g, '\n\n')    // Limit consecutive newlines
      .trim();
  }
  
  private isRetryableError(error: any): boolean {
    const retryableErrors = ['NetworkError', 'TimeoutError', 'ServiceUnavailable'];
    return retryableErrors.some(type => error.name === type || error.message.includes(type));
  }
}
```

```typescript
// packages/ui/src/services/MigrationRollback.ts
export class MigrationRollback {
  constructor(
    private storageService: any,
    private webdavService: any,
    private backupManager: BackupManager
  ) {}
  
  async rollbackMigration(migrationId: string): Promise<void> {
    try {
      // Load migration state
      const migrationState = await this.getMigrationState(migrationId);
      if (!migrationState) {
        throw new Error('Migration state not found');
      }
      
      // Restore from backup
      await this.restoreFromBackup(migrationState.backupPath);
      
      // Clean up WebDAV files if they were created
      if (migrationState.uploadedFiles && migrationState.uploadedFiles.length > 0) {
        await this.cleanupWebDAVFiles(migrationState.uploadedFiles);
      }
      
      // Reset migration state
      await this.resetMigrationState(migrationId);
      
      showNotification('Migration rolled back successfully', 'info');
      
    } catch (error) {
      console.error('Rollback failed:', error);
      showNotification('Rollback failed - manual recovery may be needed', 'error');
      throw error;
    }
  }
  
  private async restoreFromBackup(backupPath: string): Promise<void> {
    if (!backupPath) {
      throw new Error('No backup path specified');
    }
    
    const backup = await this.backupManager.loadBackup(backupPath);
    if (!backup) {
      throw new Error('Backup not found');
    }
    
    // Restore each legacy storage key
    for (const [key, data] of Object.entries(backup.data)) {
      await this.storageService.setItem(key, JSON.stringify(data));
    }
  }
  
  private async cleanupWebDAVFiles(filePaths: string[]): Promise<void> {
    const cleanupResults = await Promise.allSettled(
      filePaths.map(async (filePath) => {
        try {
          await this.webdavService.deleteFile(filePath);
        } catch (error) {
          console.warn(`Failed to cleanup file ${filePath}:`, error);
        }
      })
    );
    
    const failedCleanups = cleanupResults.filter(result => result.status === 'rejected');
    if (failedCleanups.length > 0) {
      console.warn(`${failedCleanups.length} files could not be cleaned up from WebDAV`);
    }
  }
  
  async createRestorePoint(): Promise<string> {
    const restoreId = `restore_${Date.now()}`;
    const currentState = await this.captureCurrentState();
    
    await this.storageService.setItem(
      `restore_point_${restoreId}`,
      JSON.stringify({
        id: restoreId,
        timestamp: new Date(),
        state: currentState
      })
    );
    
    return restoreId;
  }
  
  private async captureCurrentState(): Promise<any> {
    const state = {};
    
    // Capture all relevant storage keys
    const keysToCapture = [
      'prompt_migration_v2_state',
      'saved_prompts',
      'prompt_data',
      'user_prompts',
```

```typescript
// packages/ui/src/services/MigrationErrorHandler.ts
export class MigrationErrorHandler {
  handleMigrationError(error: Error, context: string, prompt?: LegacyPrompt): void {
    let errorMessage = 'Migration failed';
    let troubleshootingSteps: string[] = [];
    let isRecoverable = false;
    
    // Categorize errors and provide specific guidance
    if (error.name === 'NetworkError') {
      errorMessage = 'Network connection failed during migration';
      troubleshootingSteps = [
        'Check your internet connection',
        'Verify WebDAV server is accessible',
        'Try again when connection is stable'
      ];
      isRecoverable = true;
      
    } else if (error.message.includes('quota') || error.message.includes('space')) {
      errorMessage = 'Insufficient storage space on WebDAV server';
      troubleshootingSteps = [
        'Free up space on your WebDAV server',
        'Consider migrating in smaller batches',
        'Contact your server administrator'
      ];
      isRecoverable = true;
      
    } else if (error.message.includes('authentication')) {
      errorMessage = 'WebDAV authentication failed';
      troubleshootingSteps = [
        'Check your WebDAV credentials',
        'Verify server permissions',
        'Re-configure WebDAV connection'
      ];
      isRecoverable = true;
      
    } else if (error.message.includes('timeout')) {
      errorMessage = 'Migration timed out';
      troubleshootingSteps = [
        'Server may be slow or overloaded',
        'Try migrating fewer items at once',
        'Check server status'
      ];
      isRecoverable = true;
      
    } else if (error.message.includes('verification failed')) {
      errorMessage = 'File upload verification failed';
      troubleshootingSteps = [
        'Server may have modified the file',
        'Check server configuration',
        'Try uploading manually to test'
      ];
      isRecoverable = false;
    }
    
    // Log detailed error for debugging
    console.error(`Migration error in ${context}:`, {
      error: error.message,
      stack: error.stack,
      prompt: prompt ? { id: prompt.id, title: prompt.title } : undefined,
      context,
      timestamp: new Date()
    });
    
    // Show user-friendly notification with actions
    const actions = [];
    
    if (isRecoverable) {

        label: 'Retry',

      });
    }
    
    actions.push({
      label: 'View Details',
      action: () => this.showErrorDetails(error, troubleshootingSteps)
    });
    
    showNotification(errorMessage, 'error', 0, actions);
  }

      'warning',
```
