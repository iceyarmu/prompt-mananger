```typescript
// packages/ui/src/services/PromptExecutionHandler.ts
import { ILLMService, StreamHandlers } from '@prompt-optimizer/core';
import type { AppServices } from '../types/services';

export interface ExecutionOptions {
  model: string;
  temperature?: number;
  maxTokens?: number;
  systemPrompt?: string;
  streaming?: boolean;
  timeout?: number;
}

export interface ExecutionResult {
  content: string;
  model: string;
  usage: {
    promptTokens: number;
    completionTokens: number;
    totalTokens: number;
    cost: number;
  };
  executionTime: number;
  timestamp: Date;
  streaming?: boolean;
}

export class PromptExecutionHandler {
  constructor(
    private services: AppServices,
    private onProgress?: (content: string) => void
  ) {}
  
  async executePrompt(
    prompt: string,
    options: ExecutionOptions
  ): Promise<ExecutionResult> {
    const startTime = Date.now();
    
    try {
      // 使用现有LLM服务
      const llmService = this.services.llmService;
      
      if (options.streaming) {
        return await this.executeStreaming(llmService, prompt, options);
      }
      
      // 使用现有服务模式的标准执行
      const response = await llmService.testPrompt(
        options.systemPrompt || '',
        prompt,
        options.model
      );
      
      return {
        content: response,
        model: options.model,
        usage: {
          promptTokens: this.calculateTokens(prompt),
          completionTokens: this.calculateTokens(response),
          totalTokens: this.calculateTokens(prompt) + this.calculateTokens(response),
          cost: this.calculateCost(options.model, this.calculateTokens(prompt), this.calculateTokens(response))
        },
        executionTime: Date.now() - startTime,
        timestamp: new Date()
      };
      
    } catch (error) {
      return this.handleExecutionError(error, options);
    }
  }
  
  private async executeStreaming(
    llmService: ILLMService,
    prompt: string,
    options: ExecutionOptions
  ): Promise<ExecutionResult> {
    let content = '';
    const startTime = Date.now();
    
    const handlers: StreamHandlers = {
      onStart: () => {
        this.onProgress?.('');
      },
      onToken: (token: string) => {
        content += token;
        this.onProgress?.(content);
      },
      onFinish: () => {
        // 流式传输完成
      },
      onError: (error: Error) => {
        throw error;
      }
    };
    
    await llmService.testPromptStream(
      options.systemPrompt || '',
      prompt,
      options.model,
      handlers
    );
    
    return {
      content,
      model: options.model,
      usage: {
        promptTokens: this.calculateTokens(prompt),
        completionTokens: this.calculateTokens(content),
        totalTokens: this.calculateTokens(prompt) + this.calculateTokens(content),
        cost: this.calculateCost(options.model, this.calculateTokens(prompt), this.calculateTokens(content))
      },
      executionTime: Date.now() - startTime,
      timestamp: new Date(),
      streaming: true
    };
  }
  
  private calculateTokens(text: string): number {
    // 粗略的令牌计算 - 在生产中使用适当的令牌器
    return Math.ceil(text.length / 4);
  }
  
  private calculateCost(model: string, promptTokens: number, completionTokens: number): number {
    const pricing = this.getModelPricing(model);
    return (promptTokens * pricing.prompt + completionTokens * pricing.completion) / 1000;
  }
  
  private getModelPricing(model: string) {
    const pricingMap = {
      'gpt-4': { prompt: 0.03, completion: 0.06 },
      'gpt-3.5-turbo': { prompt: 0.001, completion: 0.002 },
      'claude-3-opus': { prompt: 0.015, completion: 0.075 },
      'claude-3-sonnet': { prompt: 0.003, completion: 0.015 },
      'claude-3-haiku': { prompt: 0.00025, completion: 0.00125 }
    };
    return pricingMap[model] || { prompt: 0, completion: 0 };
  }
}
```

```typescript
export class ExecutionErrorHandler {
  private retryDelays = [1000, 2000, 4000, 8000]; // 指数退避
  
  async handleExecutionError(
    error: Error,
    options: ExecutionOptions,
    retryCount = 0
  ): Promise<ExecutionResult> {
    let errorMessage = '执行失败';
    let shouldRetry = false;
    let retryDelay = 0;
    
    if (error.message.includes('rate limit')) {
      errorMessage = '超出速率限制';
      shouldRetry = retryCount < 3;
      retryDelay = this.retryDelays[retryCount] || 8000;
      
      if (shouldRetry) {
        showNotification(
          `速率限制。将在${retryDelay / 1000}秒后重试...`,
          'warning'
        );
        
        await new Promise(resolve => setTimeout(resolve, retryDelay));
        return this.retryExecution(options, retryCount + 1);
      }
      
    } else if (error.message.includes('timeout')) {
      errorMessage = '请求超时';
      shouldRetry = retryCount < 2;
      
    } else if (error.message.includes('quota')) {
      errorMessage = '超出API配额';
      
    } else if (error.message.includes('authentication')) {
      errorMessage = '身份验证失败 - 检查API密钥';
      
    } else if (error.message.includes('model not found')) {
      errorMessage = '选定的模型不可用';
    }
    
    const actions = [];
    if (shouldRetry) {
      actions.push({
        label: '重试',
        action: () => this.retryExecution(options, retryCount + 1)
      });
    }
    
    showNotification(errorMessage, 'error', 0, actions);
    
    throw new Error(errorMessage);
  }
```

### 使用量跟踪和分析

```typescript
// packages/ui/src/services/UsageTracker.ts
export interface UsageStats {
  totalTokens: number;
  totalCost: number;
  executionCount: number;
  averageExecutionTime: number;
  modelUsage: Record<string, {
    count: number;
    tokens: number;
    cost: number;
  }>;
}

export class UsageTracker {
  private readonly STORAGE_KEY = 'execution_usage_stats';
  
  async trackExecution(result: ExecutionResult) {
    const stats = await this.getUsageStats();
    
    stats.totalTokens += result.usage.totalTokens;
    stats.totalCost += result.usage.cost;
    stats.executionCount += 1;
    
    // 更新模型特定统计信息
    if (!stats.modelUsage[result.model]) {
      stats.modelUsage[result.model] = { count: 0, tokens: 0, cost: 0 };
    }
    
    const modelStats = stats.modelUsage[result.model];
    modelStats.count += 1;
    modelStats.tokens += result.usage.totalTokens;
    modelStats.cost += result.usage.cost;
    
    // 更新平均执行时间
    stats.averageExecutionTime = (
      (stats.averageExecutionTime * (stats.executionCount - 1) + result.executionTime)
      / stats.executionCount
    );
    
    await this.saveUsageStats(stats);
    
    // 如果支出很高则显示成本警告
    if (result.usage.cost > 0.10) { // $0.10
      showNotification(
        `高成本执行: $${result.usage.cost.toFixed(4)}`,
        'warning'
      );
    }
  }
  
  async getUsageStats(): Promise<UsageStats> {
    const stored = await storageService.getItem(this.STORAGE_KEY);
    if (!stored) {
      return {
        totalTokens: 0,
        totalCost: 0,
        executionCount: 0,
        averageExecutionTime: 0,
        modelUsage: {}
      };
    }
    return JSON.parse(stored);
  }
  
  private async saveUsageStats(stats: UsageStats) {
    await storageService.setItem(this.STORAGE_KEY, JSON.stringify(stats));
  }
}
```
