# Story 1.11: 实现迁移和向后兼容性

## 状态
草稿

## 用户故事
**作为** 用户，
**我希望** 我现有的本地提示词能够迁移到新系统，
**以便** 我不会丢失之前的工作。

## 验收标准
1. 在首次启动时检测本地存储中的现有提示词，并进行版本检查
2. 提示用户进行迁移，明确解释好处和过程
3. 在迁移前创建本地存储的备份，支持可下载的JSON导出
4. 将每个本地提示词转换为.md文件格式，并保留适当的元数据
5. 使用现有服务模式将转换的文件上传到WebDAV存储
6. 在清除本地存储前验证迁移成功，并进行完整性检查
7. 如果迁移失败，提供回滚选项，支持完整恢复功能
8. 迁移对每个浏览器/用户只运行一次，支持持久完成跟踪
9. 如果跳过迁移，现有应用保持功能正常，支持优雅降级
10. 迁移过程不阻塞UI，支持适当的异步处理和进度指示
11. 迁移期间显示进度指示，包括分步反馈和取消选项
12. 本地存储在迁移确认前保持不变，支持用户验证
13. 迁移过程中零数据丢失，支持原子操作和验证
14. 优雅处理迁移中断，支持恢复功能
15. 为迁移失败提供清晰的错误消息，包含具体故障排除步骤
16. 备份可作为JSON文件下载，支持完整数据导出包括元数据

### 遗留数据检测
[来源: 现有存储模式和数据结构]
```typescript
// packages/ui/src/services/MigrationManager.ts
interface LegacyPrompt {
  id: string;
  title: string;
  content: string;
  created: string | Date;
  modified: string | Date;
  tags?: string[];
  category?: string;
  version?: string;
  metadata?: Record<string, any>;
}

interface MigrationState {
  status: 'not_started' | 'in_progress' | 'completed' | 'failed' | 'cancelled';
  startTime?: Date;
  endTime?: Date;
  totalItems: number;
  processedItems: number;
  failedItems: number;
  currentStep: string;
  backupPath?: string;
  migrationId: string;
}

export class MigrationManager {
  private readonly MIGRATION_KEY = 'prompt_migration_v2_state';
  private readonly LEGACY_KEYS = [
    'saved_prompts',           // v1.0格式
    'prompt_data',             // v1.1格式
    'user_prompts',            // v1.2格式
    'optimization_history'     // 优化历史
  ];
  
  constructor(
    private storageService: any,
    private webdavService: any,
    private backupManager: BackupManager
  ) {}
  
  async checkMigrationNeeded(): Promise<{
    needed: boolean;
    dataFound: LegacyPrompt[];
    estimatedTime: number;
  }> {
    // 检查迁移是否已完成
    const migrationState = await this.getMigrationState();
    if (migrationState?.status === 'completed') {
      return { needed: false, dataFound: [], estimatedTime: 0 };
    }
    
    // 检测所有已知键中的遗留数据
    const legacyData: LegacyPrompt[] = [];
    
    for (const key of this.LEGACY_KEYS) {
      const data = await this.storageService.getItem(key);
      if (data) {
        try {
          const parsed = JSON.parse(data);
          const prompts = Array.isArray(parsed) ? parsed : [parsed];
          legacyData.push(...this.normalizeLegacyPrompts(prompts));
        } catch (error) {
          console.warn(`解析${key}的遗留数据失败:`, error);
        }
      }
    }
    
    // 基于内容哈希去除重复项
    const uniquePrompts = this.deduplicatePrompts(legacyData);
    
    return {
      needed: uniquePrompts.length > 0,
      dataFound: uniquePrompts,
      estimatedTime: this.estimateMigrationTime(uniquePrompts.length)
    };
  }
  
  private normalizeLegacyPrompts(rawData: any[]): LegacyPrompt[] {
    return rawData.map((item, index) => ({
      id: item.id || `legacy_${index}_${Date.now()}`,
      title: item.title || item.name || `无标题 ${index + 1}`,
      content: item.content || item.text || '',
      created: new Date(item.created || item.createdAt || Date.now()),
      modified: new Date(item.modified || item.updatedAt || item.created || Date.now()),
      tags: item.tags || [],
      category: item.category || '已迁移',
      version: item.version || '1.0',
      metadata: {
        originalId: item.id,
        migrationSource: 'localStorage',
        ...item.metadata
      }
    }));
  }
  
  private deduplicatePrompts(prompts: LegacyPrompt[]): LegacyPrompt[] {
    const seen = new Set<string>();
    return prompts.filter(prompt => {
      const hash = this.generateContentHash(prompt.content + prompt.title);
      if (seen.has(hash)) {
        return false;
      }
      seen.add(hash);
      return true;
    });
  }
  
  private generateContentHash(content: string): string {
    // 用于去重的简单哈希函数
    let hash = 0;
    for (let i = 0; i < content.length; i++) {
      const char = content.charCodeAt(i);
      hash = ((hash << 5) - hash) + char;
      hash = hash & hash; // 转换为32位整数
    }
    return hash.toString();
  }
  
  private estimateMigrationTime(promptCount: number): number {
    // 估算: 每个提示词约100ms + 开销
    return Math.max(promptCount * 100 + 2000, 5000);
  }
}
```

### 数据转换和上传服务
[来源: 现有WebDAV模式和markdown转换]
```typescript
// packages/ui/src/services/MigrationConverter.ts
export class MigrationConverter {
  constructor(
    private webdavService: any,
    private progressCallback?: (progress: number, operation: string) => void
  ) {}
  
  async convertAndUpload(
    prompts: LegacyPrompt[],
    migrationFolder: string = '/migrated'
  ): Promise<MigrationResult> {
    const results: MigrationResult = {
      successful: [],
      failed: [],
      totalProcessed: 0
    };
    
    // 创建迁移文件夹结构
    try {
      await this.webdavService.createFolder(migrationFolder);
      await this.webdavService.createFolder(`${migrationFolder}/categories`);
    } catch (error) {
      console.warn('创建迁移文件夹失败:', error);
    }
    
    // 批量转换和上传以提高性能
    const batchSize = 5;
    for (let i = 0; i < prompts.length; i += batchSize) {
      const batch = prompts.slice(i, i + batchSize);
      
      await Promise.allSettled(
        batch.map(async (prompt) => {
          try {
            const convertedContent = this.convertToMarkdown(prompt);
            const filePath = this.generateFilePath(prompt, migrationFolder);
            
            this.progressCallback?.(
              (i + batch.indexOf(prompt)) / prompts.length * 100,
              `正在上传 ${prompt.title}...`
            );
            
            await this.webdavService.putFile(filePath, convertedContent);
            
            // 验证上传
            const uploaded = await this.webdavService.getFile(filePath);
            if (uploaded !== convertedContent) {
              throw new Error('上传验证失败');
            }
            
            results.successful.push({
              original: prompt,
              filePath,
              size: convertedContent.length
            });
            
          } catch (error) {
            console.error(`迁移提示词"${prompt.title}"失败:`, error);
            results.failed.push({
              prompt,
              error: error.message,
              retryable: this.isRetryableError(error)
            });
          } finally {
            results.totalProcessed++;
          }
        })
      );
      
      // 批次之间稍作暂停以防止服务器过载
      await new Promise(resolve => setTimeout(resolve, 100));
    }
    
    return results;
  }
  
  private convertToMarkdown(prompt: LegacyPrompt): string {
    const frontmatter = this.generateFrontmatter(prompt);
    const content = this.processContent(prompt.content);
    
    return `---
${frontmatter}
---

# ${prompt.title}

${content}

---

## 迁移信息
- **原始ID**: ${prompt.id}
- **创建时间**: ${new Date(prompt.created).toISOString()}
- **修改时间**: ${new Date(prompt.modified).toISOString()}
- **迁移时间**: ${new Date().toISOString()}
- **来源**: 本地存储迁移
${prompt.tags?.length ? `- **标签**: ${prompt.tags.join(', ')}` : ''}
${prompt.category ? `- **类别**: ${prompt.category}` : ''}
`;
  }
  
  private generateFrontmatter(prompt: LegacyPrompt): string {
    const metadata = {
      title: prompt.title,
      created: new Date(prompt.created).toISOString(),
      modified: new Date(prompt.modified).toISOString(),
      migrated: new Date().toISOString(),
      tags: prompt.tags || [],
      category: prompt.category || 'migrated',
      version: prompt.version || '1.0',
      id: prompt.id,
      ...prompt.metadata
    };
    
    return Object.entries(metadata)
      .map(([key, value]) => {
        if (Array.isArray(value)) {
          return `${key}: [${value.map(v => `"${v}"`).join(', ')}]`;
        }
        return `${key}: "${value}"`;
      })
      .join('\n');
  }
  
  private generateFilePath(prompt: LegacyPrompt, basePath: string): string {
    const sanitizedTitle = this.sanitizeFilename(prompt.title);
    const category = prompt.category ? this.sanitizeFilename(prompt.category) : 'general';
    
    // 创建基于类别的文件夹结构
    const categoryPath = `${basePath}/categories/${category}`;
    const filename = `${sanitizedTitle}.md`;
    
    return `${categoryPath}/${filename}`;
  }
  
  private sanitizeFilename(name: string): string {
    return name
      .replace(/[<>:"/\\|?*]/g, '')  // 移除无效字符
      .replace(/\s+/g, '-')          // 用破折号替换空格
      .replace(/--+/g, '-')          // 用单个破折号替换多个破折号
      .replace(/^-|-$/g, '')         // 移除开头/结尾的破折号
      .toLowerCase()
      .substring(0, 100);            // 限制长度
  }
  
  private processContent(content: string): string {
    // 为markdown清理内容
    return content
      .replace(/\r\n/g, '\n')        // 标准化行结束符
      .replace(/\n{3,}/g, '\n\n')    // 限制连续换行符
      .trim();
  }
  
  private isRetryableError(error: any): boolean {
    const retryableErrors = ['NetworkError', 'TimeoutError', 'ServiceUnavailable'];
    return retryableErrors.some(type => error.name === type || error.message.includes(type));
  }
}
```

```typescript
// packages/ui/src/services/MigrationRollback.ts
export class MigrationRollback {
  constructor(
    private storageService: any,
    private webdavService: any,
    private backupManager: BackupManager
  ) {}
  
  async rollbackMigration(migrationId: string): Promise<void> {
    try {
      // 加载迁移状态
      const migrationState = await this.getMigrationState(migrationId);
      if (!migrationState) {
        throw new Error('未找到迁移状态');
      }
      
      // 从备份恢复
      await this.restoreFromBackup(migrationState.backupPath);
      
      // 如果创建了WebDAV文件则清理
      if (migrationState.uploadedFiles && migrationState.uploadedFiles.length > 0) {
        await this.cleanupWebDAVFiles(migrationState.uploadedFiles);
      }
      
      // 重置迁移状态
      await this.resetMigrationState(migrationId);
      
      showNotification('迁移已成功回滚', 'info');
      
    } catch (error) {
      console.error('回滚失败:', error);
      showNotification('回滚失败 - 可能需要手动恢复', 'error');
      throw error;
    }
  }
  
  private async restoreFromBackup(backupPath: string): Promise<void> {
    if (!backupPath) {
      throw new Error('未指定备份路径');
    }
    
    const backup = await this.backupManager.loadBackup(backupPath);
    if (!backup) {
      throw new Error('未找到备份');
    }
    
    // 恢复每个遗留存储键
    for (const [key, data] of Object.entries(backup.data)) {
      await this.storageService.setItem(key, JSON.stringify(data));
    }
  }
  
  private async cleanupWebDAVFiles(filePaths: string[]): Promise<void> {
    const cleanupResults = await Promise.allSettled(
      filePaths.map(async (filePath) => {
        try {
          await this.webdavService.deleteFile(filePath);
        } catch (error) {
          console.warn(`清理文件${filePath}失败:`, error);
        }
      })
    );
    
    const failedCleanups = cleanupResults.filter(result => result.status === 'rejected');
    if (failedCleanups.length > 0) {
      console.warn(`${failedCleanups.length}个文件无法从WebDAV清理`);
    }
  }
  
  async createRestorePoint(): Promise<string> {
    const restoreId = `restore_${Date.now()}`;
    const currentState = await this.captureCurrentState();
    
    await this.storageService.setItem(
      `restore_point_${restoreId}`,
      JSON.stringify({
        id: restoreId,
        timestamp: new Date(),
        state: currentState
      })
    );
    
    return restoreId;
  }
  
  private async captureCurrentState(): Promise<any> {
    const state = {};
    
    // 捕获所有相关存储键
    const keysToCapture = [
      'prompt_migration_v2_state',
      'saved_prompts',
      'prompt_data',
      'user_prompts',
      'optimization_history'
    ];
    
    for (const key of keysToCapture) {
      const data = await this.storageService.getItem(key);
      if (data) {
        state[key] = data;
      }
    }
    
    return state;
  }
}
```

```typescript
// packages/ui/src/services/MigrationErrorHandler.ts
export class MigrationErrorHandler {
  handleMigrationError(error: Error, context: string, prompt?: LegacyPrompt): void {
    let errorMessage = '迁移失败';
    let troubleshootingSteps: string[] = [];
    let isRecoverable = false;
    
    // 分类错误并提供具体指导
    if (error.name === 'NetworkError') {
      errorMessage = '迁移期间网络连接失败';
      troubleshootingSteps = [
        '检查您的网络连接',
        '验证WebDAV服务器是否可访问',
        '连接稳定时重试'
      ];
      isRecoverable = true;
      
    } else if (error.message.includes('quota') || error.message.includes('space')) {
      errorMessage = 'WebDAV服务器存储空间不足';
      troubleshootingSteps = [
        '在WebDAV服务器上释放空间',
        '考虑分批次迁移',
        '联系您的服务器管理员'
      ];
      isRecoverable = true;
      
    } else if (error.message.includes('authentication')) {
      errorMessage = 'WebDAV身份验证失败';
      troubleshootingSteps = [
        '检查您的WebDAV凭据',
        '验证服务器权限',
        '重新配置WebDAV连接'
      ];
      isRecoverable = true;
      
    } else if (error.message.includes('timeout')) {
      errorMessage = '迁移超时';
      troubleshootingSteps = [
        '服务器可能很慢或过载',
        '尝试一次迁移较少的项目',
        '检查服务器状态'
      ];
      isRecoverable = true;
      
    } else if (error.message.includes('verification failed')) {
      errorMessage = '文件上传验证失败';
      troubleshootingSteps = [
        '服务器可能修改了文件',
        '检查服务器配置',
        '尝试手动上传测试'
      ];
      isRecoverable = false;
    }
    
    // 记录详细错误以供调试
    console.error(`${context}中的迁移错误:`, {
      error: error.message,
      stack: error.stack,
      prompt: prompt ? { id: prompt.id, title: prompt.title } : undefined,
      context,
      timestamp: new Date()
    });
    
    // 显示用户友好的通知和操作
    const actions = [];
    
    if (isRecoverable) {
      actions.push({
        label: '重试',
        action: () => this.retryOperation(context, prompt)
      });
    }
    
    actions.push({
      label: '查看详情',
      action: () => this.showErrorDetails(error, troubleshootingSteps)
    });
    
    showNotification(errorMessage, 'error', 0, actions);
  }
  
  private showErrorDetails(error: Error, troubleshootingSteps: string[]): void {
    // 创建详细错误模态框
    const errorDetails = {
      title: '迁移错误详情',
      error: error.message,
      timestamp: new Date(),
      troubleshooting: troubleshootingSteps
    };
    
    // 使用现有模态框系统显示详情
    // 实现取决于模态框系统结构
  }
  
  handleInterruption(migrationState: MigrationState): void {
    showNotification(
      '迁移被中断。您可以从中断的地方恢复。',
      'warning',
      0,
      [{
        label: '恢复迁移',
        action: () => this.resumeMigration(migrationState.migrationId)
```
