# Story 1.7: 实现文件持久性和同步

## 状态
草稿

## 故事
**作为**用户，
**我希望**我的文件更改能够可靠地保存到WebDAV，
**以便**确保不丢失工作成果，并在设备间同步。

## 验收标准
1. 文件更改自动保存，500ms防抖延迟，带有适当的防抖和批处理
2. 手动保存选项（Ctrl+S），优先于自动保存，带有即时反馈
3. 保存状态可见指示器（已保存/保存中/未保存/错误），带有适当的UI反馈
4. 网络中断时保存队列，带有离线模式和重试机制
5. 冲突检测和解决（如果文件在其他地方被修改），带有三向合并
6. 在页面卸载时保存未保存的更改，带有beforeunload处理
7. 保存失败时显示错误通知，带有重试选项
8. 优化批量更改（避免每次按键时保存），使用智能批处理
9. 离线模式支持，本地存储作为后备，带有同步队列
10. 文件版本历史跟踪（可选），用于冲突解决和审计
11. 保存操作在2秒内完成，带有超时处理
12. 从WebDAV服务器自动重新加载外部更改，带有更改检测
13. 多标签同步（同一文件在多个标签中打开），使用广播通道
14. 内存中备份用于意外崩溃恢复，带有周期性持久化
15. 保存进度指示器用于大文件（>1MB），带有可取消操作

### 保存管理架构
[来源：现有服务模式]
```typescript
// packages/ui/src/services/SaveManager.ts
export interface SaveState {
  status: 'saved' | 'saving' | 'unsaved' | 'error' | 'offline';
  lastSaved?: Date;
  error?: string;
  hasUnsavedChanges: boolean;
  queuedChanges?: number;
}

export class SaveManager {
  private saveQueue = new Map<string, string>(); // filePath -> content
  private saveTimeouts = new Map<string, NodeJS.Timeout>();
  private savePromises = new Map<string, Promise<void>>();
  private onlineStatus = navigator.onLine;
  
  constructor(
    private webdavService: WebDAVService,
    private offlineQueue: OfflineSaveQueue,
    private conflictResolver: ConflictResolver
  ) {
    this.setupNetworkListeners();
    this.setupBeforeUnload();
  }
  
  async saveFile(filePath: string, content: string, immediate = false): Promise<void> {
    if (immediate) {
      return this.saveImmediately(filePath, content);
    }
    
    // Queue the save with debouncing
    this.queueSave(filePath, content);
  }
  
  private queueSave(filePath: string, content: string) {
    // Clear existing timeout
    if (this.saveTimeouts.has(filePath)) {
      clearTimeout(this.saveTimeouts.get(filePath)!);
    }
    
    // Update queue
    this.saveQueue.set(filePath, content);
    this.notifyStateChange(filePath, { status: 'unsaved', hasUnsavedChanges: true });
    
    // Set new timeout
    const timeout = setTimeout(() => {
      this.processSave(filePath);
    }, 500);
    
    this.saveTimeouts.set(filePath, timeout);
  }
  
  private async processSave(filePath: string): Promise<void> {
    const content = this.saveQueue.get(filePath);
    if (!content) return;
    
    try {
      this.notifyStateChange(filePath, { status: 'saving' });
      
      if (!this.onlineStatus) {
        await this.offlineQueue.enqueue(filePath, content);
        this.notifyStateChange(filePath, { status: 'offline', queuedChanges: this.offlineQueue.size });
        return;
      }
      
      // Check for conflicts
      const conflict = await this.conflictResolver.checkConflict(filePath, content);
      if (conflict) {
        await this.handleConflict(filePath, content, conflict);
        return;
      }
      
      // Save to WebDAV
      await this.saveToWebDAV(filePath, content);
      
      this.saveQueue.delete(filePath);
      this.saveTimeouts.delete(filePath);
      this.notifyStateChange(filePath, {
        status: 'saved',
        lastSaved: new Date(),
        hasUnsavedChanges: false
      });
      
    } catch (error) {
      console.error(`Save failed for ${filePath}:`, error);
      this.notifyStateChange(filePath, {
        status: 'error',
        error: error.message,
        hasUnsavedChanges: true
      });
      
      // Requeue for retry
      this.scheduleRetry(filePath, content);
    }
  }
  
  private async saveToWebDAV(filePath: string, content: string): Promise<void> {
    const controller = new AbortController();
    const timeout = setTimeout(() => controller.abort(), 2000);
    
    try {
      await this.webdavService.putFile(filePath, content, {
        signal: controller.signal,
        onProgress: (progress) => {
          if (content.length > 1024 * 1024) { // 1MB
            this.notifyProgress(filePath, progress);
          }
        }
      });
    } finally {
      clearTimeout(timeout);
    }
  }
}
```

### 离线队列实现
[来源：PWA最佳实践]
```typescript
// packages/ui/src/services/OfflineSaveQueue.ts
interface QueuedSave {
  filePath: string;
  content: string;
  timestamp: Date;
  attempts: number;
  lastAttempt?: Date;
}

export class OfflineSaveQueue {
  private queue: QueuedSave[] = [];
  private readonly STORAGE_KEY = 'offline_save_queue';
  private readonly MAX_ATTEMPTS = 5;
  private processingQueue = false;
  
  constructor(private webdavService: WebDAVService) {
    this.loadFromStorage();
    this.setupOnlineListener();
  }
  
  async enqueue(filePath: string, content: string): Promise<void> {
    const existingIndex = this.queue.findIndex(item => item.filePath === filePath);
    
    if (existingIndex >= 0) {
      // Update existing entry
      this.queue[existingIndex].content = content;
      this.queue[existingIndex].timestamp = new Date();
    } else {
      // Add new entry
      this.queue.push({
        filePath,
        content,
        timestamp: new Date(),
        attempts: 0
      });
    }
    
    await this.saveToStorage();
  }
  
  private async processQueue(): Promise<void> {
    if (this.processingQueue || !navigator.onLine) return;
    
    this.processingQueue = true;
    
    try {
      for (let i = 0; i < this.queue.length; i++) {
        const item = this.queue[i];
        
        if (item.attempts >= this.MAX_ATTEMPTS) {
          // Move to failed queue or notify user
          continue;
        }
        
        try {
          await this.webdavService.putFile(item.filePath, item.content);
          
          // Remove from queue on success
          this.queue.splice(i, 1);
          i--; // Adjust index
          
          showNotification(`Synced ${item.filePath}`, 'success');
          
        } catch (error) {
          item.attempts++;
          item.lastAttempt = new Date();
          
          // Exponential backoff
          const delay = Math.min(1000 * Math.pow(2, item.attempts), 30000);
          setTimeout(() => this.processQueue(), delay);
          
          console.warn(`Failed to sync ${item.filePath}, attempt ${item.attempts}:`, error);
        }
      }
    } finally {
      this.processingQueue = false;
      await this.saveToStorage();
    }
  }
}
```

```typescript
// packages/ui/src/services/ConflictResolver.ts
interface FileConflict {
  filePath: string;
  localContent: string;
  remoteContent: string;
  localModified: Date;
  remoteModified: Date;
  baseContent?: string; // For three-way merge
}

export class ConflictResolver {
  private fileVersions = new Map<string, string>(); // filePath -> etag/version
  
  async checkConflict(filePath: string, content: string): Promise<FileConflict | null> {
    try {
      // Get current file info from WebDAV
      const fileInfo = await this.webdavService.getFileInfo(filePath);
      const lastKnownVersion = this.fileVersions.get(filePath);
      
      if (lastKnownVersion && fileInfo.etag !== lastKnownVersion) {
        // File was modified externally
        const remoteContent = await this.webdavService.getFile(filePath);
        
        return {
          filePath,
          localContent: content,
          remoteContent,
          localModified: new Date(),
          remoteModified: fileInfo.modified,
          baseContent: this.getBaseContent(filePath)
        };
      }
      
      // Update known version
      this.fileVersions.set(filePath, fileInfo.etag);
      return null;
      
    } catch (error) {
      // File doesn't exist or network error
      return null;
    }
  }
  
  async resolveConflict(conflict: FileConflict, resolution: 'local' | 'remote' | 'merge'): Promise<string> {
    switch (resolution) {
      case 'local':
        return conflict.localContent;
        
      case 'remote':
        return conflict.remoteContent;
        
      case 'merge':
        if (conflict.baseContent) {
          return this.threeWayMerge(
            conflict.baseContent,
            conflict.localContent,
            conflict.remoteContent
          );
        } else {
          // Simple merge - concatenate with separator
          return `${conflict.localContent}\n\n--- MERGED FROM REMOTE ---\n\n${conflict.remoteContent}`;
        }
    }
  }
  
  private threeWayMerge(base: string, local: string, remote: string): string {
    // Implement three-way merge algorithm
    // This is a simplified version - in production, consider using a library like diff3
    const localDiff = this.getDiff(base, local);
    const remoteDiff = this.getDiff(base, remote);
    
    // Apply both diffs if they don't conflict
    if (!this.hasConflictingChanges(localDiff, remoteDiff)) {
      return this.applyDiffs(base, [localDiff, remoteDiff]);
    }
    
    // Return manual merge format
    return `<<<<<<< LOCAL\n${local}\n=======\n${remote}\n>>>>>>> REMOTE`;
  }
}
```

### 多标签同步
[来源：BroadcastChannel API]
```typescript
// packages/ui/src/services/TabSync.ts
export class TabSync {
  private channel: BroadcastChannel;
  private openFiles = new Set<string>();
  
  constructor() {
    this.channel = new BroadcastChannel('prompt-manager-sync');
    this.setupMessageHandler();
  }
  
  private setupMessageHandler() {
    this.channel.addEventListener('message', (event) => {
      const { type, filePath, content, timestamp } = event.data;
      
      switch (type) {
        case 'file_changed':
          this.handleFileChanged(filePath, content, timestamp);
          break;
          
        case 'file_opened':
          this.handleFileOpened(filePath);
          break;
          
        case 'file_closed':
          this.handleFileClosed(filePath);
          break;
      }
    });
  }
  
  broadcastFileChange(filePath: string, content: string) {
    this.channel.postMessage({
      type: 'file_changed',
      filePath,
      content,
      timestamp: Date.now()
    });
  }
  
  private handleFileChanged(filePath: string, content: string, timestamp: number) {
    // Update editor content if file is open in this tab
    const editorStore = useEditorStore();
    if (editorStore.currentFile === filePath) {
      // Only update if the change is newer and not from this tab
      if (timestamp > editorStore.lastModified) {
        editorStore.updateContent(content, false); // Don't trigger save
      }
    }
  }
}
```

```typescript
// packages/ui/src/services/CrashRecovery.ts
interface RecoveryData {
  filePath: string;
  content: string;
  timestamp: Date;
  sessionId: string;
}

export class CrashRecovery {
  private readonly STORAGE_KEY = 'crash_recovery_data';
  private readonly SESSION_ID = this.generateSessionId();
  private recoveryInterval: NodeJS.Timeout | null = null;
  
  constructor() {
    this.checkForRecovery();
    this.startPeriodicBackup();
  }
  
  private checkForRecovery() {
    const recoveryData = this.getRecoveryData();
    
    if (recoveryData.length > 0) {
      showNotification(
        `Found ${recoveryData.length} unsaved files from previous session`,
        'info',
        0,
        [{
          label: 'Recover',
          action: () => this.showRecoveryDialog(recoveryData)
        }, {
          label: 'Discard',
          action: () => this.clearRecoveryData()
        }]
      );
    }
  }
  
  backupContent(filePath: string, content: string) {
    const recoveryData = this.getRecoveryData();
    const existingIndex = recoveryData.findIndex(item => item.filePath === filePath);
    
    const backupItem: RecoveryData = {
      filePath,
      content,
      timestamp: new Date(),
      sessionId: this.SESSION_ID
    };
    
    if (existingIndex >= 0) {
      recoveryData[existingIndex] = backupItem;
    } else {
      recoveryData.push(backupItem);
    }
    
    localStorage.setItem(this.STORAGE_KEY, JSON.stringify(recoveryData));
  }
  
  clearBackup(filePath: string) {
    const recoveryData = this.getRecoveryData();
    const filtered = recoveryData.filter(item => item.filePath !== filePath);
    localStorage.setItem(this.STORAGE_KEY, JSON.stringify(filtered));
  }
}
```

```vue
<!-- packages/ui/src/components/SaveStatus.vue -->
<template>
  <div class="save-status flex items-center gap-2">
    <div
      class="status-indicator w-2 h-2 rounded-full transition-colors"
      :class="statusClasses"
    />
    
    <span class="text-sm text-gray-600 dark:text-gray-300">
      {{ statusText }}
    </span>
    
    <Button
      v-if="saveState.status === 'error'"
      variant="ghost"
      size="xs"
      @click="retrySave"
    >
      {{ t('save.retry') }}
    </Button>
    
    <!-- Progress bar for large files -->
    <div v-if="showProgress" class="flex items-center gap-2">
      <div class="w-20 h-1 bg-gray-200 rounded">
        <div
          class="h-full bg-blue-500 rounded transition-all duration-300"
          :style="{ width: `${progress}%` }"
        />
      </div>
      <span class="text-xs text-gray-500">{{ progress }}%</span>
    </div>
  </div>
</template>

<script setup lang="ts">
const props = defineProps<{
  saveState: SaveState;
  progress?: number;
}>();

const statusClasses = computed(() => {
  switch (props.saveState.status) {
    case 'saved':
      return 'bg-green-500';
    case 'saving':
      return 'bg-blue-500 animate-pulse';
    case 'unsaved':
      return 'bg-yellow-500';
    case 'error':
      return 'bg-red-500';
    case 'offline':
      return 'bg-gray-500';
    default:
      return 'bg-gray-300';
  }
});

const statusText = computed(() => {
  const { status, lastSaved, queuedChanges } = props.saveState;
  
  switch (status) {
    case 'saved':
      return lastSaved ? `Saved ${formatTimeAgo(lastSaved)}` : 'Saved';
    case 'saving':
      return 'Saving...';
    case 'unsaved':
      return 'Unsaved changes';
    case 'error':
      return 'Save failed';
    case 'offline':
      return queuedChanges ? `${queuedChanges} changes queued` : 'Offline';
    default:
      return '';
  }
});
</script>
```
