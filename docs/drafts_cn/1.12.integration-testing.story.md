// tests/performance/performance.test.ts
import { performance } from 'perf_hooks';

describe('性能基准测试', () => {
  beforeAll(async () => {
    await setupPerformanceEnvironment();
  });
  
  describe('文件操作性能', () => {
    test('在3秒内加载1000个文件', async () => {
      const mockFiles = generateMockFileTree(1000);
      
      const start = performance.now();
      await fileTreeStore.loadTree(mockFiles);
      const end = performance.now();
      
      const duration = end - start;
      expect(duration).toBeLessThan(3000);
      
      // 记录性能指标
      console.log(`在${duration.toFixed(2)}ms内加载了1000个文件`);
    });
    
    test('文件保存操作在500ms内完成', async () => {
      const testContent = generateTestContent(10000); // 10KB内容
      
      const start = performance.now();
      await saveManager.saveFile('/test.md', testContent);
      const end = performance.now();
      
      const duration = end - start;
      expect(duration).toBeLessThan(500);
    });
  });
  
  describe('内存使用', () => {
    test('100个文件的内存使用保持在200MB以下', async () => {
      const initialMemory = process.memoryUsage().heapUsed;
      
      // 加载100个大文件
      for (let i = 0; i < 100; i++) {
        const content = generateLargeContent(50000); // 每个50KB
        await editorStore.loadContent(`file-${i}.md`, content);
      }
      
      // 如果可用，强制垃圾收集
      if (global.gc) {
        global.gc();
      }
      
      const finalMemory = process.memoryUsage().heapUsed;
      const memoryIncreaseMB = (finalMemory - initialMemory) / 1024 / 1024;
      
      expect(memoryIncreaseMB).toBeLessThan(200);
      console.log(`内存增加: ${memoryIncreaseMB.toFixed(2)}MB`);
    });
  });
  
  describe('UI性能', () => {
    test('编辑器输入期间保持60fps', async () => {
      const editor = await mountEditor();
      const frameTimes: number[] = [];
      
      let frameCount = 0;
      const measureFrame = () => {
        const now = performance.now();
        if (frameTimes.length > 0) {
          const frameTime = now - frameTimes[frameTimes.length - 1];
          if (frameTime > 16.67) { // 60fps = 每帧16.67ms
            frameCount++;
          }
        }
        frameTimes.push(now);
        
        if (frameTimes.length < 60) {
          requestAnimationFrame(measureFrame);
        }
      };
      
      requestAnimationFrame(measureFrame);
      
      // 模拟大量输入
      await editor.simulateTyping('Lorem ipsum '.repeat(100));
      
      // 等待测量完成
      await new Promise(resolve => setTimeout(resolve, 1000));
      
      const droppedFrames = frameCount;
      const fps = 1000 / (frameTimes[frameTimes.length - 1] - frameTimes[0]) * frameTimes.length;
      
      expect(fps).toBeGreaterThan(55); // 允许一些差异
      expect(droppedFrames).toBeLessThan(5);
    });
  });
  
  describe('包大小分析', () => {
    test('初始包大小在限制内', async () => {
      const bundleStats = await analyzeBundleSize();
      
      expect(bundleStats.initialChunk).toBeLessThan(500 * 1024); // 500KB
      expect(bundleStats.totalSize).toBeLessThan(2 * 1024 * 1024); // 2MB
      
      console.log('包分析:', {
        initial: `${(bundleStats.initialChunk / 1024).toFixed(2)}KB`,
        total: `${(bundleStats.totalSize / 1024 / 1024).toFixed(2)}MB`,
        chunks: bundleStats.chunkCount
      });
    });
  });
});

  private setupPerformanceMonitoring() {
    // Core Web Vitals监控
    if ('PerformanceObserver' in window) {
      this.performanceObserver = new PerformanceObserver((list) => {
        list.getEntries().forEach((entry) => {
          if (entry.entryType === 'largest-contentful-paint') {
            this.trackMetric('LCP', entry.startTime);
          } else if (entry.entryType === 'first-input') {
            this.trackMetric('FID', entry.processingStart - entry.startTime);
          } else if (entry.entryType === 'layout-shift') {
            if (!entry.hadRecentInput) {
              this.trackMetric('CLS', entry.value);
            }
          }
        });
      });
      
      this.performanceObserver.observe({
        entryTypes: ['largest-contentful-paint', 'first-input', 'layout-shift']
      });
    }
    
    // 自定义性能监控
    this.monitorFileOperations();
    this.monitorMemoryUsage();
  }
  
  private setupErrorTracking() {
    // 全局错误处理器
    window.addEventListener('error', (event) => {
      this.trackError(event.error, {
        filename: event.filename,
        lineno: event.lineno,
        colno: event.colno,
        type: 'javascript'
      });
    });
    
    // Promise拒绝处理器
    window.addEventListener('unhandledrejection', (event) => {
      this.trackError(event.reason, {
        type: 'promise-rejection'
      });
    });
    
    // Vue错误处理器
    if (window.Vue) {
      window.Vue.config.errorHandler = (error, vm, info) => {
        this.trackError(error, {
          component: vm?.$options.name,
          errorInfo: info,
          type: 'vue'
        });
      };
    }
  }
  
  private setupAnalytics() {
    // 用户行为跟踪
    this.trackPageViews();
    this.trackFeatureUsage();
    this.trackUserJourney();
  }
  
  trackFileOperation(operation: string, duration: number, success: boolean) {
    const metric = {
      name: 'file_operation',
      operation,
      duration,
      success,
      timestamp: Date.now()
    };
    
    // 发送到分析
    this.sendMetric(metric);
    
    // 如果性能下降则发出警报
    if (duration > 500) {
      this.sendAlert('slow_file_operation', {
        operation,
        duration,
        threshold: 500
      });
    }
  }
  
  trackMemoryUsage() {
    if ('memory' in performance) {
      const usage = (performance as any).memory.usedJSHeapSize / 1024 / 1024;
      
      this.sendMetric({
        name: 'memory_usage',
        value: usage,
        timestamp: Date.now()
      });
      
      if (usage > 180) {
        this.sendAlert('high_memory_usage', {
          usage,
          threshold: 180
        });
      }
    }
  }
  
  trackUserEngagement(feature: string, action: string, metadata?: any) {
    this.sendMetric({
      name: 'user_engagement',
      feature,
      action,
      metadata,
      timestamp: Date.now(),
      sessionId: this.getSessionId()
    });
  }
  
  private sendMetric(metric: any) {
    // 发送到您的分析服务（如Google Analytics、Mixpanel）
    if (window.gtag) {
      window.gtag('event', metric.name, metric);
    }
    
    // 在开发中也记录调试
    if (process.env.NODE_ENV === 'development') {
      console.log('📊 指标:', metric);
    }
  }
  
  private sendAlert(alertType: string, data: any) {
    // 发送到监控服务（如Sentry、DataDog）
    console.warn(`🚨 警报: ${alertType}`, data);
    
    // 在生产中，发送到您的监控服务
    // 示例: Sentry.captureMessage(`Alert: ${alertType}`, 'warning', { extra: data });
  }
}

// packages/ui/src/services/FeatureFlagService.ts
export interface FeatureFlag {
  key: string;
  enabled: boolean;
  rolloutPercentage: number;
  targetGroups?: string[];
  conditions?: FeatureFlagCondition[];
}

export interface FeatureFlagCondition {
  type: 'user_attribute' | 'browser' | 'date_range' | 'custom';
  operator: 'equals' | 'contains' | 'greater_than' | 'less_than';
  value: any;
}

export class FeatureFlagService {
  private flags: Map<string, FeatureFlag> = new Map();
  private userId: string;
  
  constructor() {
    this.userId = this.getUserId();
    this.loadFeatureFlags();
  }
  
  private async loadFeatureFlags() {
    try {
      // 在生产中，从功能标志服务加载
      const response = await fetch('/api/feature-flags', {
        headers: {
          'X-User-ID': this.userId
        }
      });
      
      const flags = await response.json();
      flags.forEach((flag: FeatureFlag) => {
        this.flags.set(flag.key, flag);
      });
    } catch (error) {
      console.warn('加载功能标志失败:', error);
      // 回退到默认标志
      this.loadDefaultFlags();
    }
  }
  
  private loadDefaultFlags() {
    const defaultFlags: FeatureFlag[] = [
      {
        key: 'webdav_integration',
        enabled: true,
        rolloutPercentage: 100
      },
      {
        key: 'advanced_editor',
        enabled: true,
        rolloutPercentage: 80
      },
      {
        key: 'ai_execution',
        enabled: true,
        rolloutPercentage: 50,
        targetGroups: ['beta_users', 'power_users']
      },
      {
        key: 'migration_wizard',
        enabled: true,
        rolloutPercentage: 25
      }
    ];
    
    defaultFlags.forEach(flag => {
      this.flags.set(flag.key, flag);
    });
  }
  
  isEnabled(flagKey: string): boolean {
    const flag = this.flags.get(flagKey);
    if (!flag) {
      console.warn(`未找到功能标志: ${flagKey}`);
      return false;
    }
    
    // 检查是否全局禁用
    if (!flag.enabled) {
      return false;
    }
    
    // 检查发布百分比
    const userHash = this.hashUserId(this.userId);
    const userPercentile = (userHash % 100) + 1;
    
    if (userPercentile > flag.rolloutPercentage) {
      return false;
    }
    
    // 检查目标组
    if (flag.targetGroups && flag.targetGroups.length > 0) {
      const userGroups = this.getUserGroups();
      const hasTargetGroup = flag.targetGroups.some(group => 
        userGroups.includes(group)
      );
      
      if (!hasTargetGroup) {
        return false;
      }
    }
    
    // 检查条件
    if (flag.conditions && flag.conditions.length > 0) {
      return flag.conditions.every(condition => 
        this.evaluateCondition(condition)
      );
    }
    
    return true;
  }
  
  private hashUserId(userId: string): number {
    let hash = 0;
    for (let i = 0; i < userId.length; i++) {
      const char = userId.charCodeAt(i);
      hash = ((hash << 5) - hash) + char;
      hash = hash & hash; // 转换为32位整数
    }
    return Math.abs(hash);
  }
  
  // 在组件中的使用
  static createDirective() {
    return {
      mounted(el: HTMLElement, binding: { value: string }) {
        const flagService = new FeatureFlagService();
        if (!flagService.isEnabled(binding.value)) {
          el.style.display = 'none';
        }
      }
    };
  }
}

// Vue组合式API
export function useFeatureFlags() {
  const flagService = new FeatureFlagService();
  
  return {
    isEnabled: (flagKey: string) => flagService.isEnabled(flagKey),
    whenEnabled: (flagKey: string, callback: () => void) => {
      if (flagService.isEnabled(flagKey)) {
        callback();
      }
    }
  };
}

  const report: QualityReport = {

    lintErrors,
    typeErrors,
    performanceScore,
    accessibilityScore,
    bundleSize: bundleSize / 1024 / 1024 // 转换为MB
  };
  
  // 质量门
